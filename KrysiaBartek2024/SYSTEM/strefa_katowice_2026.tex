                \item 1\spades = \trsf{\nt/\clubs}

                \item 2\spades = \clubs, preempt/\inv/\gf\ (\then\ 2\nt = accept, 3\clubs = weak)
                \item 2\nt = \minor, weak/\gf
                \item 3\clubs = \mixed
                \item 3\diams = preempt
                \item 3\hearts/3\spades = splinter (weaker than through 2\spades)
            \end{itemize}

        \section*{\colorbox{blue!30}{Opponents' overcalls over 1\hearts\ opening}}
        \addcontentsline{toc}{section}{Opponents' overcalls over 1\hearts\ opening} {

            \subsubsection*{1\hearts -- (\dbl) -- ?}
            \begin{itemize}
                \item 1\spades = \nat\ 4+\spades
                \item 1\nt = \trsf{\clubs}
                \item 2\clubs = \trsf{\diams}
                \item 2\diams = \hearts\ constructive+
                \item 2\hearts = \hearts\ weak
                \item 2\spades = \nat\ preempt
                \item 2\nt = \invp, (3)4+\hearts
                \item 3\clubs = \mixed
                \item 3\diams = mini splinter \gf
                \item 3\hearts = preempt
                \item 3\spades/4\clubs/4\diams = splinter
                \item 4\spades = to play
            \end{itemize}

            \subsubsection*{1\hearts -- (1\spades) -- ?}
            \begin{itemize}
                \item \dbl = takeout
                \item 1\nt = \trsf{\clubs}
                \item 2\clubs = \trsf{\diams}
                \item 2\diams = \hearts\ constructive+
                \item 2\hearts = \hearts\ weak
                \item 2\spades = \invp\ \small{OFF}
                \item 2\nt = \invp\ \small{DEF}
                \item 3\clubs = \mixed
                \item 3\diams = mini splinter \gf
                \item 3\spades = splinter
                \item 4\clubs/4\diams = color + fit
                \item 4\spades = exclusion
            \end{itemize}

            \subsubsection*{1\hearts -- (1\nt) -- ?}
            \begin{itemize}
                \item \dbl = penalty
                \item other = \nat
            \end{itemize}

            \subsubsection*{1\hearts -- (2\clubs) -- ?}
            \begin{itemize}
                \item 2\diams/2\hearts/2\spades = \nat\ \nf
                \item 2\nt = \invp, fit
                \item 3\clubs = \mixed
                \item 3\diams/3\spades = \nat\ \gf
                \item 3\hearts = preempt
            \end{itemize}

            \vspace{0.3cm}
            General rule: level 2 = \nf, level 3 = \gf. Of 2\nt\ and the cue-bid, the lower = \invp\ (fit), the higher = \mixed.

        
        }

        \section*{\colorbox{blue!30}{Opponents' overcalls over 1\ntx\ opening}}
        \addcontentsline{toc}{section}{Opponents' overcalls over 1\ntx\ opening} {

            \subsubsection*{1\ntx\ -- (\alrts{2\clubs/2\diams}) -- ?}
            2\clubs/2\diams\ = \major
            \begin{itemize}
                \item \dbl = penalty
                \item 2\diams/2\hearts = \nat\ \nf
                \item 2\spades = \minor, weak/\gf
                \item 2\nt = \leb
                \item 3\clubs/3\diams/3\hearts = transfer, \invp
                \item 3\spades = \trsf{\clubs}, \gf
                \item 3\nt = to play, w/o stopper
            \end{itemize}

            \subsubsection*{1\ntx\ -- (2\clubs/2\diams) -- 2\nt -- (P)\\
                            3\clubs -- (P) -- ?}
            \begin{itemize}
                \item \pass/3\diams = \soff
                \item 3\hearts/3\spades = ask stopper
                \item 3\nt = to play, with stopper
            \end{itemize}

            \subsubsection*{1\ntx\ -- (2\clubs) -- ?}
            2\clubs\ = \nat
            \begin{itemize}
                \item \dbl = Stayman
                \item other = as without interference
            \end{itemize}

            \subsubsection*{1\ntx\ -- (2\diams) -- ?}
            2\diams\ = \nat
            \begin{itemize}
                \item \dbl = takeout (may be weak!)
                \item 2\hearts/2\spades = \nat\ \nf
                \item 2\nt = \leb
                \item 3\diams/3\hearts = transfer, \invp
                \item 3\clubs = ask stopper
            \end{itemize}

            \subsubsection*{1\ntx\ -- (\alrts{2\diams}) -- ?}
            2\diams\ = multi
            \begin{itemize}
                \item \dbl = strength (subsequent \dbl = penalty)
                \item 2\hearts/2\spades = \nat\ \nf
                \item 2\nt = \leb
                \item 3\clubs/3\diams/3\hearts = transfer, \invp
                \item \pass\ then opponents' suit = ask stopper
            \end{itemize}

            \subsubsection*{1\ntx\ -- (2\hearts) -- ?}
            2\hearts\ = \nat\ or \hearts\ + other suit
            \begin{itemize}
                \item \dbl = takeout (may be weak!)
                \item 2\spades = \nat
                \item 2\nt = \leb
                \item 3\clubs/3\hearts/3\spades = transfer, \invp
                \item 3\diams = ask stopper
                \item 3\nt = to play, w/o stopper
                \item 4\hearts = Texas
                \item 4\spades = to play
            \end{itemize}

            \subsubsection*{1\ntx\ -- (2\spades) -- ?}
            2\spades\ = \nat\ or \spades\ + other suit
            \begin{itemize}
                \item \dbl = takeout (may be weak!)
                \item 2\nt = \leb
                \item 3\clubs/3\diams/3\spades = transfer, \invp
                \item 3\hearts = ask stopper
                \item 3\nt = to play, w/o stopper
                \item 4\diams = Texas
                \item 4\hearts = to play
            \end{itemize}

            \subsubsection*{1\ntx\ -- (2\nt) -- ?}
            2\nt\ = \minor
            \begin{itemize}
                \item \dbl = penalty
                \item 3\clubs = \invp\ \hearts\
                \item 3\diams = \invp\ \spades
                \item 3\hearts/3\spades = to play
            \end{itemize}

            \subsubsection*{1\ntx\ -- (3\clubs) -- ?}
            \begin{itemize}
                \item \dbl = takeout \gf
                \item 3\diams = \trsf{\hearts}, \invp
                \item 3\hearts = \trsf{\spades}, \invp
                \item 3\spades = \trsf{\diams}, \invp
                \item 3\nt = to play
                \item 4\diams/4\hearts = Texas
            \end{itemize}

            \subsubsection*{1\ntx\ -- (3\diams) -- ?}
            \begin{itemize}
                \item \dbl = takeout \gf
                \item 3\hearts = \trsf{\spades}, \invp
                \item 3\spades = \trsf{\hearts}, \gf
                \item 3\nt = to play
                \item 4\diams/4\hearts = Texas
            \end{itemize}

            \subsubsection*{1\ntx\ -- (3\hearts) -- ?}
            \begin{itemize}
                \item \dbl = \trsf{\spades}, \invp \vimp
                \item 3\spades = takeout \gf \vimp
                \item 3\nt = to play
                \item 4\hearts = Texas
                \item 4\spades = to play
            \end{itemize}

            \subsubsection*{1\ntx\ -- (3\spades) -- ?}
            \begin{itemize}
                \item \dbl = takeout \gf
            \end{itemize}
